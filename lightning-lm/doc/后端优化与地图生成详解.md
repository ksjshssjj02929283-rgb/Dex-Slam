### 1.3 代码实现流程

#### 1.3.1 初始化

```cpp
void LoopClosing::Init(const std::string yaml_path) {
    // 1. 配置优化器
    miao::OptimizerConfig config(
        miao::AlgorithmType::LEVENBERG_MARQUARDT,    // LM算法
        miao::LinearSolverType::LINEAR_SOLVER_SPARSE_EIGEN,  // 稀疏求解器
        false                                         // 不启用详细输出
    );
    config.incremental_mode_ = true;  // 关键：启用增量模式
    optimizer_ = miao::SetupOptimizer<6, 3>(config);
    
    // 2. 设置运动约束信息矩阵（相邻关键帧）
    info_motion_.setIdentity();
    info_motion_.block<3, 3>(0, 0) = 
        Mat3d::Identity() / (motion_trans_noise_² );  // 位移信息矩阵
    info_motion_.block<3, 3>(3, 3) = 
        Mat3d::Identity() / (motion_rot_noise_²);     // 旋转信息矩阵
    
    // 3. 设置回环约束信息矩阵
    info_loops_.setIdentity();
    info_loops_.block<3, 3>(0, 0) = 
        Mat3d::Identity() / (loop_trans_noise_²);
    info_loops_.block<3, 3>(3, 3) = 
        Mat3d::Identity() / (loop_rot_noise_²);
    
    // 4. 启动异步处理线程（在线模式）
    if (options_.online_mode_) {
        kf_thread_.SetProcFunc([this](Keyframe::Ptr kf) { 
            HandleKF(kf); 
        });
        kf_thread_.Start();
    }
}
```

**关键参数**：
- `motion_trans_noise_ = 0.1m` - 运动约束位移标准差
- `motion_rot_noise_ = 3°` - 运动约束旋转标准差
- `loop_trans_noise_ = 0.2m` - 回环约束位移标准差
- `loop_rot_noise_ = 3°` - 回环约束旋转标准差

信息矩阵 = 协方差矩阵的逆 = 1/σ²

## 2. 约束构建详解

### 2.1 约束类型总览

Lightning-LM的后端使用两种约束：

| 约束类型 | 作用 | 连接关系 | 信息矩阵 |
|---------|------|---------|---------|
| **运动约束** | 保持轨迹连续性 | 相邻关键帧 | `info_motion_` |
| **回环约束** | 消除累积误差 | 回环关键帧对 | `info_loops_` |

### 2.2 回环候选检测

#### 2.2.1 检测条件

```cpp
void LoopClosing::DetectLoopCandidates() {
    candidates_.clear();
    
    // 条件1：时间间隔检查
    // 每隔 loop_kf_gap 个关键帧才检测一次
    if (last_loop_kf_ && 
        (cur_kf_->GetID() - last_loop_kf_->GetID()) <= loop_kf_gap_) {
        return;  // 时间间隔太近，跳过
    }
    
    // 条件2：遍历历史关键帧
    for (auto kf : all_keyframes_) {
        // 条件2.1：同一轨迹内的ID间隔
        if (check_first != nullptr && 
            abs(kf->GetID() - check_first->GetID()) <= min_id_interval_) {
            continue;  // 与上一个候选帧太近
        }
        
        // 条件2.2：与当前帧的ID间隔
        if (abs(kf->GetID() - cur_kf_->GetID()) < closest_id_th_) {
            break;  // 时间上太接近，不可能是回环
        }
        
        // 条件3：空间距离检查（XY平面）
        Vec3d dt = kf->GetOptPose().translation() - 
                   cur_kf_->GetOptPose().translation();
        double t2d = dt.head<2>().norm();
        
        if (t2d < max_range_) {
            // 通过所有检查，添加为候选
            LoopCandidate c(kf->GetID(), cur_kf_->GetID());
            c.Tij_ = kf->GetLIOPose().inverse() * cur_kf_->GetLIOPose();
            candidates_.emplace_back(c);
            check_first = kf;
        }
    }
}
```

**检测条件总结**：
1. **时间条件**：`cur_id - last_loop_id > loop_kf_gap (20帧)`
2. **ID间隔**：`|cur_id - hist_id| > closest_id_th (50帧)`
3. **候选间隔**：`|cand1_id - cand2_id| > min_id_interval (20帧)`
4. **空间条件**：`XY平面距离 < max_range (30m)`

### 2.3 回环位姿计算

#### 2.3.1 子地图构建

```cpp
void LoopClosing::ComputeForCandidate(LoopCandidate& c) {
    const int submap_idx_range = 40;  // 前后各40帧
    
    auto build_submap = [](int given_id, bool build_in_world) -> CloudPtr {
        CloudPtr submap(new PointCloudType);
        
        // 聚合前后40帧，每4帧取一次
        for (int idx = -submap_idx_range; idx < submap_idx_range; idx += 4) {
            int id = idx + given_id;
            if (id < 0 || id > all_keyframes_.size()) continue;
            
            auto kf = all_keyframes_[id];
            CloudPtr cloud = kf->GetCloud();
            
            // 变换到世界坐标系或局部坐标系
            SE3 Twb = kf->GetLIOPose();
            if (!build_in_world) {
                Twb = all_keyframes_[given_id]->GetLIOPose().inverse() * Twb;
            }
            
            CloudPtr cloud_trans(new PointCloudType);
            pcl::transformPointCloud(*cloud, *cloud_trans, Twb.matrix());
            *submap += *cloud_trans;
        }
        return submap;
    };
    
    // 历史帧：构建世界坐标系下的子地图（目标点云）
    auto submap_kf1 = build_submap(kf1->GetID(), true);
    
    // 当前帧：单帧点云（源点云）
    CloudPtr submap_kf2 = kf2->GetCloud();
}
```

**子地图特点**：
- 历史帧：前后80帧范围，每4帧采样，约20帧聚合
- 当前帧：仅单帧点云
- 目的：增加历史帧的点云密度，提高匹配成功率

#### 2.3.2 多分辨率NDT配准

```cpp
void LoopClosing::ComputeForCandidate(LoopCandidate& c) {
    // ... 子地图构建 ...
    
    Mat4f Tw2 = kf2->GetLIOPose().matrix().cast<float>();
    CloudPtr output(new PointCloudType);
    
    // 多分辨率配准：从粗到精
    std::vector<double> res{10.0, 5.0, 2.0, 1.0};
    
    for (auto& r : res) {
        // 配置NDT
        pcl::NormalDistributionsTransform<PointType, PointType> ndt;
        ndt.setTransformationEpsilon(0.05);  // 变换收敛阈值
        ndt.setStepSize(0.7);                // 牛顿法步长
        ndt.setMaximumIterations(40);        // 最大迭代次数
        ndt.setResolution(r);                // 当前分辨率
        
        // 降采样（减少计算量）
        rough_map1 = VoxelGrid(submap_kf1, r * 0.1);
        rough_map2 = VoxelGrid(submap_kf2, r * 0.1);
        
        // 设置点云
        ndt.setInputTarget(rough_map1);  // 目标（历史帧子地图）
        ndt.setInputSource(rough_map2);  // 源（当前帧）
        
        // 执行配准
        ndt.align(*output, Tw2);
        
        // 获取结果作为下一个分辨率的初值
        Tw2 = ndt.getFinalTransformation();
        c.ndt_score_ = ndt.getTransformationProbability();
    }
    
    // 计算相对位姿 T_i^j = T_w^i^(-1) * T_w^j
    Mat4d T = Tw2.cast<double>();
    c.Tij_ = kf1->GetLIOPose().inverse() * SE3(T);
}
```

**多分辨率策略**：
1. **第1轮（10m）**: 粗配准，快速收敛到大致位置
2. **第2轮（5m）**: 中等精度
3. **第3轮（2m）**: 较高精度
4. **第4轮（1m）**: 精细配准，最终结果

**NDT匹配得分**：
```cpp
score = ndt.getTransformationProbability();
// score 越小越好，阈值 ndt_score_th = 1.0
```

#### 2.3.3 候选筛选

```cpp
void LoopClosing::ComputeLoopCandidates() {
    // 并行计算所有候选
    std::for_each(candidates_.begin(), candidates_.end(), 
        [this](LoopCandidate& c) { 
            ComputeForCandidate(c); 
        });
    
    // 筛选成功的候选
    std::vector<LoopCandidate> succ_candidates;
    for (const auto& lc : candidates_) {
        if (lc.ndt_score_ > ndt_score_th_) {  // 通过阈值检查
            succ_candidates.emplace_back(lc);
        }
    }
    
    candidates_.swap(succ_candidates);
}
```

### 2.4 位姿图构建

#### 2.4.1 图节点（Vertex）

```cpp
void LoopClosing::PoseOptimization() {
    // 为当前关键帧创建节点
    auto v = std::make_shared<miao::VertexSE3>();
    v->SetId(cur_kf_->GetID());           // 节点ID = 关键帧ID
    v->SetEstimate(cur_kf_->GetOptPose()); // 初始估计 = 当前优化位姿
    
    optimizer_->AddVertex(v);
    kf_vert_.emplace_back(v);
}
```

**节点表示**：
- 每个关键帧对应一个SE3顶点
- 估计值：6自由度位姿 `[x, y, z, roll, pitch, yaw]`
- 优化变量：6维增量 `Δx = [Δt, Δω]`

**更新方程**：
```cpp
void VertexSE3::OplusImpl(const double *update) {
    // 平移：直接加法
    estimate_.translation() += Vec3d(update[0], update[1], update[2]);
    
    // 旋转：SO3的左乘更新
    estimate_.so3() = estimate_.so3() * SO3::exp(Vec3d(update[3], update[4], update[5]));
}
```

#### 2.4.2 运动约束（Motion Edge）

```cpp
void LoopClosing::PoseOptimization() {
    // 添加与前几个关键帧的运动约束
    for (int i = 1; i < 3; i++) {  // 前2帧
        int id = cur_kf_->GetID() - i;
        if (id >= 0) {
            auto last_kf = all_keyframes_[id];
            
            // 创建二元边
            auto e = std::make_shared<miao::EdgeSE3>();
            e->SetVertex(0, optimizer_->GetVertex(last_kf->GetID()));
            e->SetVertex(1, v);  // 当前帧顶点
            
            // 观测：从LIO得到的相对位姿
            SE3 motion = last_kf->GetLIOPose().inverse() * cur_kf_->GetLIOPose();
            e->SetMeasurement(motion);
            
            // 信息矩阵：运动约束
            e->SetInformation(info_motion_);
            
            optimizer_->AddEdge(e);
        }
    }
}
```

**运动约束特点**：
- 连接：第 `i-1` 和第 `i` 帧（也连接第 `i-2` 帧）
- 观测：`T_meas = T_{i-1}^{-1} * T_i` （LIO估计的相对位姿）
- 权重：由 `info_motion_` 决定，较小（相对运动不确定性较高）

**运动约束误差模型**：
```cpp
class EdgeSE3 : public BaseBinaryEdge<6, SE3, VertexSE3, VertexSE3> {
    void ComputeError() override {
        SE3 from = vertices_[0]->Estimate();  // T_i
        SE3 to = vertices_[1]->Estimate();    // T_j
        
        // 计算实际相对位姿
        SE3 T_ij = from.inverse() * to;
        
        // 误差 = 观测^(-1) * 实际
        SE3 delta = measurement_.inverse() * T_ij;
        
        // 误差向量：[平移误差, 旋转误差]
        error_.head<3>() = delta.translation();
        error_.tail<3>() = delta.so3().unit_quaternion().coeffs().head<3>();
    }
};
```

**误差推导**：
```
观测: T_ij_meas (measurement)
估计: T_ij_est = T_i^(-1) * T_j

误差李代数:
  δ = log(T_ij_meas^(-1) * T_ij_est)
    = log(T_ij_meas^(-1) * T_i^(-1) * T_j)

线性化后:
  e = [t_err, ω_err]^T
```

#### 2.4.3 回环约束（Loop Edge）

```cpp
void LoopClosing::PoseOptimization() {
    // 为每个成功的回环候选添加约束
    for (auto& c : candidates_) {
        auto e = std::make_shared<miao::EdgeSE3>();
        
        // 连接回环帧对
        e->SetVertex(0, optimizer_->GetVertex(c.idx1_));  // 历史帧
        e->SetVertex(1, optimizer_->GetVertex(c.idx2_));  // 当前帧
        
        // 观测：从NDT配准得到的相对位姿
        e->SetMeasurement(c.Tij_);
        
        // 信息矩阵：回环约束（权重更大）
        e->SetInformation(info_loops_);
        
        // 鲁棒核函数：抑制异常回环
        auto rk = std::make_shared<miao::RobustKernelCauchy>();
        rk->SetDelta(rk_loop_th_);  // 阈值 = 1.04
        e->SetRobustKernel(rk);
        
        optimizer_->AddEdge(e);
        edge_loops_.emplace_back(e);
    }
}
```

**回环约束特点**：
- 连接：任意两个回环帧（可能相隔很远）
- 观测：`T_meas = T_i^{-1} * T_j` （NDT配准结果）
- 权重：由 `info_loops_` 决定，较大（NDT配准精度较高）
- 鲁棒核：Cauchy核，抑制错误回环的影响

**鲁棒核函数**：
```cpp
// Cauchy核
ρ(e²) = δ² * log(1 + e²/δ²)

// 当 e² > δ² 时，权重下降
// δ = rk_loop_th = 1.04

// 实际效果：
// e² < δ²  → 正常权重
// e² > δ²  → 权重逐渐降低，抑制异常值影响
```

### 2.5 图优化求解

#### 2.5.1 优化过程

```cpp
void LoopClosing::PoseOptimization() {
    // ... 构建图 ...
    
    // 初始化优化
    optimizer_->InitializeOptimization();
    optimizer_->SetVerbose(false);
    
    // 执行优化（最多20次迭代）
    optimizer_->Optimize(20);
}
```

**优化算法**：Levenberg-Marquardt (LM)

**目标函数**：
```
min Σ ||e_motion||²_Ω_motion + Σ ρ(||e_loop||²_Ω_loop)

其中:
- e_motion: 运动约束误差
- e_loop: 回环约束误差
- Ω_motion: 运动约束信息矩阵
- Ω_loop: 回环约束信息矩阵
- ρ(): Cauchy鲁棒核函数
```

**LM算法步骤**：
```
1. 初始化 λ = λ_0
2. For iter = 1 to max_iterations:
   a. 计算所有误差 e_i
   b. 计算雅可比矩阵 J_i
   c. 构建H矩阵: H = J^T Ω J + λI
   d. 求解: H Δx = -J^T Ω e
   e. 更新: x ← x ⊕ Δx
   f. 如果误差下降，减小λ；否则增大λ
   g. 检查收敛条件
```

#### 2.5.2 异常值检测

```cpp
void LoopClosing::PoseOptimization() {
    // ... 执行优化 ...
    
    // 检测并移除异常回环约束
    int cnt_outliers = 0;
    for (auto& e : edge_loops_) {
        if (e->GetRobustKernel() == nullptr) continue;
        
        // Chi2检验：如果误差超过阈值
        if (e->Chi2() > e->GetRobustKernel()->Delta()) {
            e->SetLevel(1);  // 标记为异常值，下次优化不使用
            cnt_outliers++;
        } else {
            e->SetRobustKernel(nullptr);  // 移除鲁棒核，成为正常约束
        }
    }
    
    LOG(INFO) << "loop outliers: " << cnt_outliers << "/" << edge_loops_.size();
}
```

**Chi2统计量**：
```cpp
chi2 = e^T Ω e

// 对于6维SE3误差
// 自由度 = 6
// 阈值 delta = 1.04 对应约95%置信度
```

### 2.6 增量式优化

Lightning-LM使用**增量式优化**，关键特点：

```cpp
config.incremental_mode_ = true;  // 启用增量模式
```

**增量式 vs 批量式**：

| 特性 | 批量式 | 增量式 |
|------|--------|--------|
| 重建图 | 每次重建 | 保持图结构 |
| 计算量 | O(n³) | O(k³), k<<n |
| 内存 | 重复分配 | 复用 |
| 适用 | 离线 | 在线实时 |

**增量式原理**：
- 保留上一次的H矩阵分解结果
- 新增节点/边时，仅更新受影响的部分
- 使用滑动窗口或固定老节点策略

## 3. 地图生成流程

### 3.1 地图类型

Lightning-LM生成三种地图：

1. **LIO地图** - 前端增量式地图
2. **优化后地图** - 后端优化后的全局一致地图

### 3.2 前端地图（IVox增量地图）

#### 3.2.1 IVox地图更新

```cpp
void LaserMapping::MapIncremental() {
    PointVector points_to_add;
    PointVector point_no_need_downsample;
    
    // 遍历降采样后的当前帧
    std::for_each(index.begin(), index.end(), [&](const size_t &i) {
        // 1. 转换到世界坐标系
        PointBodyToWorld(scan_down_body_->points[i], 
                        scan_down_world_->points[i]);
        
        PointType &point_world = scan_down_world_->points[i];
        
        // 2. 判断是否需要添加到地图
        if (!nearest_points_[i].empty() && flg_EKF_inited_) {
            const PointVector &points_near = nearest_points_[i];
            
            // 计算点所在体素中心
            Eigen::Vector3f center = 
                ((point_world.getVector3fMap() / filter_size_map_min_)
                .array().floor() + 0.5) * filter_size_map_min_;
            
            // 检查最近邻点是否在同一体素中心附近
            Eigen::Vector3f dis_2_center = points_near[0].getVector3fMap() - center;
            
            if (fabs(dis_2_center.x()) > 0.5 * filter_size_map_min_ &&
                fabs(dis_2_center.y()) > 0.5 * filter_size_map_min_ &&
                fabs(dis_2_center.z()) > 0.5 * filter_size_map_min_) {
                // 最近邻不在同一体素，直接添加
                point_no_need_downsample.emplace_back(point_world);
                return;
            }
            
            // 3. 检查是否需要降采样
            bool need_add = true;
            float dist = calc_dist(point_world.getVector3fMap(), center);
            
            if (points_near.size() >= NUM_MATCH_POINTS) {
                for (int readd_i = 0; readd_i < NUM_MATCH_POINTS; readd_i++) {
                    if (calc_dist(points_near[readd_i].getVector3fMap(), center) 
                        < dist + 1e-6) {
                        need_add = false;  // 已有更近的点
                        break;
                    }
                }
            }
            
            if (need_add) {
                points_to_add.emplace_back(point_world);
            }
        } else {
            // 首次或无邻近点，直接添加
            points_to_add.emplace_back(point_world);
        }
    });
    
    // 4. 批量添加到IVox
    ivox_->AddPoints(points_to_add);
    ivox_->AddPoints(point_no_need_downsample);
}
```

**IVox特点**：
- **增量式**：O(1)插入，不需要重建
- **体素化**：空间分块，快速查询
- **自动降采样**：每个体素保留代表点
- **动态扩展**：根据机器人运动自动扩展

```cpp
CloudPtr LaserMapping::GetGlobalMap(bool use_lio_pose, 
                                    bool use_voxel, 
                                    float res) {
    CloudPtr global_map(new PointCloudType);
    pcl::VoxelGrid<PointType> voxel;
    voxel.setLeafSize(res, res, res);
    
    // 遍历所有关键帧
    for (auto &kf : all_keyframes_) {
        CloudPtr cloud = kf->GetCloud();
        CloudPtr cloud_filter(new PointCloudType);
        
        // 可选降采样
        if (use_voxel) {
            voxel.setInputCloud(cloud);
            voxel.filter(*cloud_filter);
        } else {
            cloud_filter = cloud;
        }
        
        // 变换到世界坐标系
        CloudPtr cloud_trans(new PointCloudType);
        if (use_lio_pose) {
            pcl::transformPointCloud(*cloud_filter, *cloud_trans, 
                                    kf->GetLIOPose().matrix());
        } else {
            // 使用优化后的位姿
            pcl::transformPointCloud(*cloud_filter, *cloud_trans, 
                                    kf->GetOptPose().matrix());
        }
        
        *global_map += *cloud_trans;
    }
    
    // 全局降采样
    CloudPtr global_map_filtered(new PointCloudType);
    if (use_voxel) {
        voxel.setInputCloud(global_map);
        voxel.filter(*global_map_filtered);
    }
    
    return global_map_filtered;
}
```

## 4. 优化库 miao 详解

### 4.1 miao vs g2o

**miao** 是Lightning-LM自带的轻量级图优化库，特点：

| 特性 | miao | g2o |
|------|------|-----|
| 代码量 | 小（几千行） | 大（数万行） |
| 依赖 | 仅Eigen | Eigen + 其他 |
| 增量优化 | 原生支持 | 需额外实现 |
| 学习曲线 | 平缓 | 陡峭 |
| 性能 | 优秀 | 优秀 |
| 适用场景 | 实时SLAM | 通用优化 |

```cpp
class BaseVertex {

// SE3顶点实现
class VertexSE3 : public BaseVertex<6, SE3> {
    void OplusImpl(const double *update) override {
        // 更新公式
        estimate_.translation() += Vec3d(update[0], update[1], update[2]);
        estimate_.so3() = estimate_.so3() * 
                         SO3::exp(Vec3d(update[3], update[4], update[5]));
```

#### 4.2.2 Edge (边)

```cpp
template<int D, typename E, typename VertexType1, typename VertexType2>
class BaseBinaryEdge {
    std::array<Vertex*, 2> vertices_;  // 连接的两个顶点
    E measurement_;                     // 观测值
    MatXd information_;                 // 信息矩阵
    VecX error_;                        // 误差
    RobustKernel* robust_kernel_;      // 鲁棒核
    
    // 纯虚函数：子类实现
    virtual void ComputeError() = 0;
};

// SE3边实现
class EdgeSE3 : public BaseBinaryEdge<6, SE3, VertexSE3, VertexSE3> {
    void ComputeError() override {
        SE3 from = vertices_[0]->Estimate();
        SE3 to = vertices_[1]->Estimate();
        SE3 T_ij = from.inverse() * to;
        SE3 delta = measurement_.inverse() * T_ij;
        
        error_.head<3>() = delta.translation();
        error_.tail<3>() = delta.so3().unit_quaternion().coeffs().head<3>();
    }
};
```

### 4.4 鲁棒核函数

#### 4.4.1 Cauchy核

```cpp
class RobustKernelCauchy : public RobustKernel {
    double delta_;  // 阈值
    
    double Rho(double e2) const override {
        // ρ(e²) = δ² * log(1 + e²/δ²)
        return delta_ * delta_ * log(1 + e2 / (delta_ * delta_));
    }
    
    double Weight(double e2) const override {
        // ω(e²) = δ² / (δ² + e²)
        return delta_ * delta_ / (delta_ * delta_ + e2);
    }
};
```

**效果**：
```
权重 ω(e²)
  1.0 |___
      |   \___
  0.5 |       \___
      |           \___
  0.0 |_______________\___
      0   δ²   2δ²   3δ²   e²

当 e² > δ² 时，权重快速下降
→ 异常值的影响被抑制
```

## 6. 优化效果分析

### 6.1 优化前后对比

**优化前（仅LIO）**：
- 累积误差：随时间线性增长
- 闭环误差：可能数米甚至数十米
- 轨迹漂移：明显

**优化后（LIO + 回环）**：
- 累积误差：被回环约束消除
- 闭环误差：< 0.5m（取决于NDT精度）
- 轨迹漂移：大幅减小

**计算性能**：
- 回环检测：< 1s/次（包含NDT配准）
- 位姿图优化：< 100ms/次（20次迭代，100帧）
- 地图生成：< 5s（1000帧关键帧）

**精度指标**：
- LIO轨迹精度：< 1% 行驶距离
- 闭环后精度：< 0.5m（绝对位置）
- 回环检测成功率：> 90%（良好环境）
